cmake_minimum_required(VERSION 3.22)
project(testpayload C)
add_library(testdep SHARED testdep.c)
add_library(testnative SHARED testnative.c)
target_compile_definitions(testdep PRIVATE TP_VERSION=${TP_VERSION})
target_link_libraries(testnative testdep log)
foreach(t testdep testnative)
  target_link_options(${t} PRIVATE "-Wl,-z,max-page-size=16384")
endforeach()
